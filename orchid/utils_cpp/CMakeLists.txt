cmake_minimum_required(VERSION 3.24)
project(utils_cpp LANGUAGES CXX)

find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
# Also support pybind11 installed into the selected Python environment by pip.
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
                OUTPUT_VARIABLE pybind11_cmake_dir OUTPUT_STRIP_TRAILING_WHITESPACE
                ERROR_QUIET)
find_package(pybind11 CONFIG REQUIRED HINTS "${pybind11_cmake_dir}")

pybind11_add_module(_cluster_legals MODULE src/cluster_legals_pybind.cpp)
target_compile_features(_cluster_legals PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(_cluster_legals PRIVATE /O2)
else()
    target_compile_options(_cluster_legals PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3>)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i.86)$")
        target_compile_options(_cluster_legals PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-mavx2>)
    endif()
endif()

option(UTILS_CPP_NO_OPENMP "Disable OpenMP" OFF)
if(NOT UTILS_CPP_NO_OPENMP AND NOT "$ENV{UTILS_CPP_NO_OPENMP}" STREQUAL "1")
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_link_libraries(_cluster_legals PRIVATE OpenMP::OpenMP_CXX)
endif()

set(UTILS_CPP_ENABLE_GPU "AUTO" CACHE STRING "CUDA support: AUTO, ON, or OFF")
set_property(CACHE UTILS_CPP_ENABLE_GPU PROPERTY STRINGS AUTO ON OFF)
if("$ENV{UTILS_CPP_NO_GPU}" STREQUAL "1" OR "$ENV{UTILS_CPP_ENABLE_GPU}" STREQUAL "0")
    set(UTILS_CPP_ENABLE_GPU OFF)
elseif("$ENV{UTILS_CPP_ENABLE_GPU}" STREQUAL "1")
    set(UTILS_CPP_ENABLE_GPU ON)
endif()
if(NOT UTILS_CPP_ENABLE_GPU STREQUAL "OFF")
    if(NOT CMAKE_CUDA_COMPILER)
        if(DEFINED ENV{NVCC})
            set(CMAKE_CUDA_COMPILER "$ENV{NVCC}" CACHE FILEPATH "CUDA compiler" FORCE)
        elseif(DEFINED ENV{CUDA_HOME})
            find_program(utils_cpp_nvcc nvcc HINTS "$ENV{CUDA_HOME}/bin")
            if(utils_cpp_nvcc)
                set(CMAKE_CUDA_COMPILER "${utils_cpp_nvcc}" CACHE FILEPATH "CUDA compiler" FORCE)
            endif()
        endif()
    endif()
    include(CheckLanguage)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
        enable_language(CUDA)
        target_sources(_cluster_legals PRIVATE src/cluster_legals_gpu_kernel.cu)
        target_compile_definitions(_cluster_legals PRIVATE ENABLE_GPU=1)
        target_include_directories(_cluster_legals PRIVATE ${CMAKE_CUDA_TOOLKIT_INCLUDE_DIRECTORIES})
        if("${CMAKE_CUDA_ARCHITECTURES}" STREQUAL "")
            set(CMAKE_CUDA_ARCHITECTURES native)
        endif()
        set_target_properties(_cluster_legals PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
                              CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
    elseif(UTILS_CPP_ENABLE_GPU STREQUAL "ON")
        message(FATAL_ERROR "UTILS_CPP_ENABLE_GPU=ON requires a CUDA compiler (nvcc)")
    endif()
endif()

install(TARGETS _cluster_legals LIBRARY DESTINATION utils_cpp RUNTIME DESTINATION utils_cpp)
install(DIRECTORY utils_cpp DESTINATION . FILES_MATCHING PATTERN "*.py")
